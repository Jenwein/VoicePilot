    AIResult AIServiceWrapper::CallPythonFunction(const std::string& functionName, 
                                                const nlohmann::json& args)
    {
        try
        {
            py::object pyFunction = m_AIServiceModule.attr(functionName.c_str());
            py::object pyResult;

            // 根据参数调用Python函数
            if (args.empty())
            {
                pyResult = pyFunction();
            }
            else
            {
                py::dict pyArgs;
                for (auto& [key, value] : args.items())
                {
                    if (value.is_string())
                    {
                        pyArgs[key.c_str()] = value.get<std::string>();
                    }
                    else if (value.is_number_integer())
                    {
                        pyArgs[key.c_str()] = value.get<int>();
                    }
                    else if (value.is_number_float())
                    {
                        pyArgs[key.c_str()] = value.get<double>();
                    }
                    else if (value.is_boolean())
                    {
                        pyArgs[key.c_str()] = value.get<bool>();
                    }
                    // 其他类型可以根据需要扩展
                }

                // 使用关键字参数调用
                pyResult = pyFunction(**pyArgs);
            }

            return ConvertPythonResult(pyResult);
        }
        catch (const py::error_already_set& e)
        {
            AIResult result;
            result.success = false;
            result.error_type = "Python Error";
            result.error_details = e.what();
            SetLastError("Python function call failed: " + std::string(e.what()));
            return result;
        }
        catch (const std::exception& e)
        {
            AIResult result;
            result.success = false;
            result.error_type = "Function Call Error";
            result.error_details = e.what();
            SetLastError("Function call failed: " + std::string(e.what()));
            return result;
        }
    }

    AIResult AIServiceWrapper::ConvertPythonResult(const py::object& pyResult)
    {
        AIResult result;

        try
        {
            // 将Python字典转换为C++可用的格式
            py::dict pyDict = pyResult.cast<py::dict>();

            // 检查是否成功
            if (pyDict.contains("status"))
            {
                std::string status = pyDict["status"].cast<std::string>();
                result.success = (status == "success" || status == "continue" || status == "finished");
            }
            else if (pyDict.contains("error"))
            {
                result.success = false;
                result.error_type = pyDict["error"].cast<std::string>();
                
                if (pyDict.contains("details"))
                {
                    result.error_details = pyDict["details"].cast<std::string>();
                }
            }
            else
            {
                // 如果没有明确的状态或错误字段，假设成功
                result.success = true;
            }

            // 转换整个字典为JSON
            std::string jsonStr = py::str(pyResult).cast<std::string>();
            
            // 尝试解析为JSON（需要处理Python字典格式到JSON格式的转换）
            try
            {
                // 简单的方法：使用Python的json模块进行转换
                py::module json_module = py::module::import("json");
                py::object jsonObj = json_module.attr("dumps")(pyResult);
                std::string jsonString = jsonObj.cast<std::string>();
                result.data = nlohmann::json::parse(jsonString);
            }
            catch (...)
            {
                // 如果JSON转换失败，至少保留原始数据的关键信息
                if (pyDict.contains("transcript"))
                {
                    result.data["transcript"] = pyDict["transcript"].cast<std::string>();
                }
                if (pyDict.contains("function_calls"))
                {
                    // 处理function_calls的转换会比较复杂，这里先简化处理
                    result.data["function_calls"] = "complex_data";
                }
                if (pyDict.contains("response_text"))
                {
                    result.data["response_text"] = pyDict["response_text"].cast<std::string>();
                }
                if (pyDict.contains("message"))
                {
                    result.data["message"] = pyDict["message"].cast<std::string>();
                }
            }

        }
        catch (const std::exception& e)
        {
            result.success = false;
            result.error_type = "Conversion Error";
            result.error_details = "Failed to convert Python result: " + std::string(e.what());
            SetLastError(result.error_details);
        }

        return result;
    }

bool AgentCore::ProcessUserRequestWithTools(const std::string& userRequest, std::string& finalResponse)
{
    const int MAX_ITERATIONS = 5; // 防止无限循环
    int iteration = 0;
    
    std::string currentRequest = userRequest;
    std::string previousTurn = m_ConversationHistory;
    
    while (iteration < MAX_ITERATIONS)
    {
        iteration++;
        std::cout << "[AgentCore] LLM iteration " << iteration << std::endl;
        
        // 调用LLM处理用户请求
        AIResult llmResult = m_AIServiceWrapper->ProcessUserRequest(
            currentRequest, 
            m_ToolDefsFilePath, 
            previousTurn
        );
        
        if (!llmResult.IsSuccess())
        {
            std::cerr << "[AgentCore] LLM processing failed: " << llmResult.GetErrorMessage() << std::endl;
            return false;
        }
        
        // 检查LLM返回的状态
        std::string status;
        if (llmResult.data.contains("status"))
        {
            status = llmResult.data["status"].get<std::string>();
        }
        
        std::cout << "[AgentCore] LLM status: " << status << std::endl;
        
        if (status == "finished")
        {
            // 对话完成，获取最终响应
            if (llmResult.data.contains("response_text"))
            {
                finalResponse = llmResult.data["response_text"].get<std::string>();
                std::cout << "[AgentCore] Final response: \"" << finalResponse << "\"" << std::endl;
                
                // 更新对话历史
                m_ConversationHistory = llmResult.data.dump();
                return true;
            }
            else
            {
                std::cerr << "[AgentCore] LLM finished but no response text found." << std::endl;
                return false;
            }
        }
        else if (status == "continue")
        {
            // 需要执行工具调用
            if (llmResult.data.contains("function_calls"))
            {
                std::string toolResults;
                if (!ExecuteToolCalls(llmResult.data["function_calls"], toolResults))
                {
                    std::cerr << "[AgentCore] Tool execution failed." << std::endl;
                    return false;
                }
                
                // 将工具执行结果作为下一轮的输入
                currentRequest = "Tool execution results: " + toolResults;
                previousTurn = llmResult.data.dump();
                
                std::cout << "[AgentCore] Tool execution completed, continuing conversation..." << std::endl;
            }
            else
            {
                std::cerr << "[AgentCore] LLM status is 'continue' but no function calls found." << std::endl;
                return false;
            }
        }
        else
        {
            std::cerr << "[AgentCore] Unknown LLM status: " << status << std::endl;
            return false;
        }
    }
    
    std::cerr << "[AgentCore] Maximum iterations reached, conversation terminated." << std::endl;
    return false;
}

bool AgentCore::ExecuteToolCalls(const nlohmann::json& functionCalls, std::string& toolResults)
{
    std::cout << "[AgentCore] Executing tool calls..." << std::endl;
    
    nlohmann::json results = nlohmann::json::array();
    
    try
    {
        // 处理函数调用数组
        for (const auto& call : functionCalls)
        {
            if (!call.contains("name") || !call.contains("args"))
            {
                std::cerr << "[AgentCore] Invalid function call format." << std::endl;
                continue;
            }
            
            std::string toolName = call["name"].get<std::string>();
            nlohmann::json parameters = call["args"];
            
            std::cout << "[AgentCore] Executing tool: " << toolName << std::endl;
            
            // 执行单个工具
            std::string result = ExecuteSingleTool(toolName, parameters);
            
            // 构建结果
            nlohmann::json callResult;
            callResult["tool_name"] = toolName;
            callResult["result"] = result;
            results.push_back(callResult);
            
            std::cout << "[AgentCore] Tool '" << toolName << "' result: " << result << std::endl;
        }
        
        toolResults = results.dump();
        return true;
    }
    catch (const std::exception& e)
    {
        std::cerr << "[AgentCore] Tool execution error: " << e.what() << std::endl;
        toolResults = "Tool execution failed: " + std::string(e.what());
        return false;
    }
}

std::string AgentCore::ExecuteSingleTool(const std::string& toolName, const nlohmann::json& parameters)
{
    try
    {
        auto& registry = ToolRegistry::GetInstance();
        
        if (!registry.HasTool(toolName))
        {
            return "Error: Tool '" + toolName + "' not found.";
        }
        
        // 执行工具并获取结果
        std::string result = registry.ExecuteTool(toolName, parameters);
        
        return result;
    }
    catch (const std::exception& e)
    {
        return "Error executing tool '" + toolName + "': " + std::string(e.what());
    }
}

    Returns:
        dict: 包含处理结果的字典

            失败: {"error": "错误类型", "details": "错误详情"}
    """

    try:
        # 1. 加载工具定义
        tools_definitions = []
        if tools_file and os.path.exists(tools_file):

                    tools_definitions = tools_data['tools']
                else:
                    # 其他情况，尝试直接使用（可能是单个工具定义）
